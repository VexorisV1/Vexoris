# React + TypeScript + Vite

This template provides a minimal setup to get React working in Vite with HMR and some Oxlint rules.

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

## Expanding the Oxlint configuration

If you are developing a production application, we recommend enabling type-aware lint rules by installing `oxlint-tsgolint` and editing `.oxlintrc.json`:

```json
{
  "$schema": "./node_modules/oxlint/configuration_schema.json",
  "plugins": ["react", "typescript", "oxc"],
  "options": {
    "typeAware": true
  },
  "rules": {
    "react/rules-of-hooks": "error",
    "react/only-export-components": ["warn", { "allowConstantExport": true }]
  }
}
```

See the [Oxlint rules documentation](https://oxc.rs/docs/guide/usage/linter/rules) for the full list of rules and categories.

function CursorGlow() {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const h = (e: MouseEvent) => {
      if (ref.current) {
        ref.current.style.left = `${e.clientX}px`;
        ref.current.style.top = `${e.clientY}px`;
      }
    };
    window.addEventListener('mousemove', h, { passive: true });
    return () => window.removeEventListener('mousemove', h);
  }, []);
  return <div ref={ref} className="cursor-glow" />;
}

import { createContext, useContext, useEffect, useState } from 'react';
import type { ReactNode } from 'react';
import { supabase } from './supabase';
import type { Session, User } from '@supabase/supabase-js';

export type Profile = {
  id: string;
  username: string | null;
  display_name: string | null;
  email: string | null;
  role: string;
  is_active: boolean;
  department: string | null;
  password_changed: boolean;
  mfa_setup_done: boolean;
};

type AuthContextValue = {
  session: Session | null;
  user: User | null;
  profile: Profile | null;
  loading: boolean;
  signOut: () => Promise<void>;
  refreshProfile: () => Promise<void>;
};

const AuthContext = createContext<AuthContextValue>({
  session: null, user: null, profile: null, loading: true,
  signOut: async () => {},
  refreshProfile: async () => {},
});

export function AuthProvider({ children }: { children: ReactNode }) {
  const [session, setSession] = useState<Session | null>(null);
  const [profile, setProfile] = useState<Profile | null>(null);
  const [loading, setLoading] = useState(true);

  async function loadProfile(userId: string) {
    const { data } = await supabase
      .from('profiles')
      .select('id, username, display_name, email, role, is_active, department, password_changed, mfa_setup_done')
      .eq('id', userId)
      .single();
    setProfile((data as Profile) ?? null);
    setLoading(false);
  }

  useEffect(() => {
    supabase.auth.getSession().then(({ data }) => {
      setSession(data.session);
      if (data.session) void loadProfile(data.session.user.id);
      else setLoading(false);
    });

    const { data: sub } = supabase.auth.onAuthStateChange((_e, s) => {
      setSession(s);
      if (s) void loadProfile(s.user.id);
      else { setProfile(null); setLoading(false); }
    });

    return () => sub.subscription.unsubscribe();
  }, []);

  async function signOut() {
    await supabase.auth.signOut();
    setSession(null);
    setProfile(null);
  }

  async function refreshProfile() {
    if (session) await loadProfile(session.user.id);
  }

  return (
    <AuthContext.Provider value={{ session, user: session?.user ?? null, profile, loading, signOut, refreshProfile }}>
      {children}
    </AuthContext.Provider>
  );
}

export function useAuth() { return useContext(AuthContext); }

import { supabase } from './supabase';
import type { OverviewStats, MonthlyTrendPoint, QualityKpis, DateType } from './types';

async function callRpc<T>(name: string, args: Record<string, unknown>, timeoutMs = 35_000): Promise<T> {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  try {
    const { data, error } = await supabase.rpc(name, args).abortSignal(controller.signal);
    if (error) throw new Error(`${name}: ${error.message}`);
    return data as T;
  } catch (err) {
    if (err instanceof Error && err.name === 'AbortError') {
      throw new Error(`${name}: timeout after ${timeoutMs}ms`);
    }
    throw err;
  } finally {
    clearTimeout(timer);
  }
}

export const rpc = {
  overviewStats: (p_start: string | null, p_end: string | null, p_date_type: DateType) =>
    callRpc<OverviewStats>('get_overview_stats', { p_start, p_end, p_date_type }, 60_000),

  monthlyTrend: (p_months_back = 12, p_date_type: DateType = 'discharge', p_mode = 'units') =>
    callRpc<MonthlyTrendPoint[]>('get_monthly_trend', { p_months_back, p_date_type, p_mode }),

  qualityKpis: (p_start: string | null, p_end: string | null, p_date_type: DateType) =>
    callRpc<QualityKpis>('get_quality_kpis', { p_start, p_end, p_date_type }),
};

=== web/src/hooks/useDashboard.ts ===
import { useQuery } from '@tanstack/react-query';
import { rpc } from '@/core/rpc';
import type { DateType } from '@/core/types';

const DASH_CACHE = 5 * 60 * 1000;

export function useOverviewStats(dateType: DateType = 'discharge') {
  return useQuery({
    queryKey: ['dashboard', 'overview', dateType],
    queryFn: () => rpc.overviewStats(null, null, dateType),
    staleTime: DASH_CACHE,
    retry: 1,
  });
}

export function useMonthlyTrend(monthsBack = 12, dateType: DateType = 'discharge') {
  return useQuery({
    queryKey: ['dashboard', 'trend', monthsBack, dateType],
    queryFn: () => rpc.monthlyTrend(monthsBack, dateType),
    staleTime: DASH_CACHE,
  });
}

export function useQualityKpis(dateType: DateType = 'discharge') {
  return useQuery({
    queryKey: ['dashboard', 'quality', dateType],
    queryFn: () => rpc.qualityKpis(null, null, dateType),
    staleTime: DASH_CACHE,
  });
}

import { useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { FullPageLoading } from '@/shared/components/Loading';
import { supabase } from '@/core/supabase';

export default function AuthCallbackPage() {
  const navigate = useNavigate();

  useEffect(() => {
    (async () => {
      try {
        // 1. Handle PKCE code exchange (OAuth / email confirmation links)
        const params = new URLSearchParams(window.location.search);
        const code = params.get('code');
        if (code) {
          const { error: exchErr } = await supabase.auth.exchangeCodeForSession(code);
          if (exchErr) throw exchErr;
        }

        // 2. Now read the session
        const { data, error } = await supabase.auth.getSession();
        if (error || !data.session) {
          navigate('/login', { replace: true });
          return;
        }

        // 3. Check onboarding state
        const { data: profile } = await supabase
          .from('profiles')
          .select('password_changed, mfa_setup_done')
          .eq('id', data.session.user.id)
          .single();

        if (profile && !profile.password_changed) {
          navigate('/auth/setup-account', { replace: true });
          return;
        }
        if (profile && !profile.mfa_setup_done) {
          navigate('/auth/mfa-setup', { replace: true });
          return;
        }
        navigate('/dashboard', { replace: true });
      } catch (err) {
        console.error('Auth callback error:', err);
        navigate('/login', { replace: true });
      }
    })();
  }, [navigate]);

  return <FullPageLoading label="جارٍ إتمام التحقق..." />;
}

import { Navigate, Outlet, useLocation } from 'react-router-dom';
import { useAuth } from '@/core/auth';
import { FullPageLoading } from '@/shared/components/Loading';

interface ProtectedRouteProps {
  allowedRoles?: string[];
}

export function ProtectedRoute({ allowedRoles }: ProtectedRouteProps = {}) {
  const { user, profile, loading } = useAuth();
  const location = useLocation();

  if (loading) {
    return <FullPageLoading label="جارٍ التحقق من الجلسة..." />;
  }

  if (!user) {
    return <Navigate to="/login" state={{ from: location }} replace />;
  }

  if (allowedRoles && profile && !allowedRoles.includes(profile.role)) {
    return <Navigate to="/dashboard" replace />;
  }

  return <Outlet />;
}

function strength(pass: string) {
  let s = 0;
  if (pass.length >= 8) s++;
  if (/[A-Z]/.test(pass)) s++;
  if (/[0-9]/.test(pass)) s++;
  if (/[^A-Za-z0-9]/.test(pass)) s++;
  const cfg = [
    { label: 'ضعيفة جداً', color: 'var(--danger)' }, { label: 'ضعيفة', color: 'var(--warning)' },
    { label: 'متوسطة', color: 'var(--info)' }, { label: 'قوية', color: 'var(--primary)' },
  ];
  return { score: s, ...(cfg[Math.max(0, Math.min(s - 1, 3))]) };
}

import { useState } from 'react';
import { Minus, Plus } from 'lucide-react';

const MIN_SIZE = 12;
const MAX_SIZE = 20;
const STEP = 2;

function applyFontSize(px: number) {
  document.documentElement.style.setProperty('--base-font-size', `${px}px`);
  localStorage.setItem('vex-font-size', String(px));
}

export function FontSizeControl() {
  const [size, setSize] = useState(() => {
    const saved = Number(localStorage.getItem('vex-font-size') || 14);
    return Math.max(MIN_SIZE, Math.min(MAX_SIZE, saved));
  });

  function change(delta: number) {
    const next = Math.max(MIN_SIZE, Math.min(MAX_SIZE, size + delta));
    setSize(next);
    applyFontSize(next);
  }

  const canDecrease = size > MIN_SIZE;
  const canIncrease = size < MAX_SIZE;

  return (
    <div
      className="flex items-center h-7 rounded-lg border overflow-hidden transition-colors"
      style={{
        background: 'var(--surface-2)',
        borderColor: 'var(--line)',
      }}
    >
      {/* Aa label */}
      <span
        className="px-1.5 text-[10px] font-bold select-none"
        style={{ color: 'var(--ink-3)' }}
        title={`حجم الخط الحالي: ${size}px`}
      >
        Aa
      </span>

      <div className="w-px h-4" style={{ background: 'var(--line)' }} />

      {/* Decrease */}
      <button
        type="button"
        onClick={() => change(-STEP)}
        disabled={!canDecrease}
        className="w-6 h-full flex items-center justify-center transition-colors disabled:opacity-30 disabled:cursor-not-allowed"
        style={{ color: 'var(--ink-2)' }}
        onMouseEnter={(e) => {
          if (canDecrease) (e.currentTarget as HTMLElement).style.background = 'var(--surface-3)';
        }}
        onMouseLeave={(e) => {
          (e.currentTarget as HTMLElement).style.background = 'transparent';
        }}
        title="تصغير الخط"
        aria-label="تصغير الخط"
      >
        <Minus size={11} strokeWidth={3} />
      </button>

      <div className="w-px h-4" style={{ background: 'var(--line)' }} />

      {/* Increase */}
      <button
        type="button"
        onClick={() => change(STEP)}
        disabled={!canIncrease}
        className="w-6 h-full flex items-center justify-center transition-colors disabled:opacity-30 disabled:cursor-not-allowed"
        style={{ color: 'var(--ink-2)' }}
        onMouseEnter={(e) => {
          if (canIncrease) (e.currentTarget as HTMLElement).style.background = 'var(--surface-3)';
        }}
        onMouseLeave={(e) => {
          (e.currentTarget as HTMLElement).style.background = 'transparent';
        }}
        title="تكبير الخط"
        aria-label="تكبير الخط"
      >
        <Plus size={11} strokeWidth={3} />
      </button>
    </div>
  );
}

export function initFontSize() {
  const saved = Number(localStorage.getItem('vex-font-size') || 14);
  applyFontSize(saved);
}

type Theme = 'light' | 'dark';

function applyTheme(theme: Theme) {
  document.documentElement.classList.toggle('dark', theme === 'dark');
  localStorage.setItem('vex-theme', theme);
}

export function ThemeToggle() {
  const [theme, setTheme] = useState<Theme>(() => {
    const saved = localStorage.getItem('vex-theme') as Theme | null;
    if (saved) return saved;
    return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
  });

  useEffect(() => { applyTheme(theme); }, [theme]);

  return (
    <button
      type="button"
      className="btn-icon overflow-hidden relative"
      aria-label="تبديل الوضع"
      title={theme === 'dark' ? 'الوضع النهاري' : 'الوضع الليلي'}
      onClick={() => setTheme(theme === 'dark' ? 'light' : 'dark')}
    >
      <AnimatePresence mode="wait" initial={false}>
        <motion.span
          key={theme}
          initial={{ y: -20, opacity: 0, rotate: -90 }}
          animate={{ y: 0, opacity: 1, rotate: 0 }}
          exit={{ y: 20, opacity: 0, rotate: 90 }}
          transition={{ duration: 0.25, ease: [0.16, 1, 0.3, 1] }}
          className="absolute inset-0 flex items-center justify-center"
        >
          {theme === 'dark' ? <Sun size={15} /> : <Moon size={15} />}
        </motion.span>
      </AnimatePresence>
    </button>
  );
}

import { useState } from 'react';

import { Footer } from '@/shared/components/Footer';

export function WorkspaceLayout() {
  const [mobileOpen, setMobileOpen] = useState(false);

  // Persist collapse across refreshes
  const [collapsed, setCollapsed] = useState<boolean>(() => {
    return localStorage.getItem('vex-sidebar-collapsed') === 'true';
  });

  function toggleCollapsed() {
    setCollapsed(prev => {
      const next = !prev;
      localStorage.setItem('vex-sidebar-collapsed', String(next));
      return next;
    });
  }

  return (
    <div
      className="app-shell"
      data-sidebar={collapsed ? 'collapsed' : undefined}
    >
      <Header
        variant="workspace"
        onMenuClick={() => setMobileOpen(true)}
        onCollapseToggle={toggleCollapsed}
        sidebarCollapsed={collapsed}
      />
      <Sidebar
        mobileOpen={mobileOpen}
        onMobileClose={() => setMobileOpen(false)}
        collapsed={collapsed}
        onCollapseToggle={toggleCollapsed}
      />
      <div
        className="sidebar-overlay"
        data-open={mobileOpen ? 'true' : 'false'}
        onClick={() => setMobileOpen(false)}
      />
      <main className="app-main">
        <div className="content-container">
          <Outlet />
        </div>
      </main>
      <Footer variant="workspace" />

export const fmtMonth = (iso: string): string => {
  const [y, m] = iso.split('-');
  const date = new Date(Number(y), Number(m) - 1, 1);
